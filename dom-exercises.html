2 Get Form Values

Solution

const form = document.getElementById('ex2-form');

  e.preventDefault();

  const table = document.getElementById('ex5-table');

6 Update Table Cell

Nhận row, column và chuỗi mới rồi cập nhật nội dung ô tương ứng.

Row1 cell1Row1 cell2
Row2 cell1Row2 cell2
Row3 cell1Row3 cell2

Solution

function changeContent(row, col, value) {
  const table = document.getElementById('ex6-table');
  if (row < table.rows.length && col < table.rows[row].cells.length) {
    table.rows[row].cells[col].textContent = value;
    return true;
  }
  return false;
}

7 Create Table Dynamically

Nhập số hàng và cột để tạo bảng; mỗi ô hiển thị "Row-i Column-j".

Solution

function createTable(rows, cols) {
  const table = document.createElement('table');
  for (let i = 0; i < rows; i++) {
    const tr = table.insertRow(i);
    for (let j = 0; j < cols; j++) {
      const td = tr.insertCell(j);
      td.textContent = `Row-${i} Column-${j}`;
    }
  }
  const holder = document.getElementById('ex7-holder');
  holder.innerHTML = '';
  holder.appendChild(table);
}

11 Random Image Display

Bấm nút để hiển thị một ảnh ngẫu nhiên từ danh sách.

Solution

const images = [
  { src: 'https://picsum.photos/id/1015/240/160', width: 240, height: 160 },
  { src: 'https://picsum.photos/id/1025/320/195', width: 320, height: 195 },
  { src: 'https://picsum.photos/id/1035/500/343', width: 500, height: 343 },
];
document.getElementById('ex11-btn').addEventListener('click', () => {
  const pick = images[Math.floor(Math.random() * images.length)];
  const img = document.createElement('img');
  img.src = pick.src;
  img.width = pick.width;
  img.height = pick.height;
  img.alt = 'Random';
  const holder = document.getElementById('ex11-holder');
  holder.innerHTML = '';
  holder.appendChild(img);
});

12 Highlight Bold on Hover

Khi rê chuột lên link, tô sáng các từ in đậm trong đoạn văn.

[Rê chuột vào đây để tô sáng các từ in đậm]

We have just started this section for the users (beginner to intermediate) who want to work with various JavaScript problems and write scripts online to test their JavaScript skill.

Solution

const link = document.getElementById('ex12-link');
const bolds = document.querySelectorAll('#ex12-para b');
link.addEventListener('mouseover', () => {
  bolds.forEach((b) => { b.style.backgroundColor = '#fde047'; });
});
link.addEventListener('mouseout', () => {
  bolds.forEach((b) => { b.style.backgroundColor = ''; });
});

13 Window Resize Dimensions

Hiển thị chiều rộng/cao cửa sổ và cập nhật mỗi khi resize.

Thử thay đổi kích thước cửa sổ trình duyệt.

Solution

function showSize() {
  document.getElementById('ex13-out').textContent =
